<script>
/* ═══════════════════════════════════════════════
   叶序水滴场
   第 n 颗的角度 = 黄金角 137.5° × n，半径 = √n。
   每颗单独成形、单独淡入 —— 一颗一颗慢慢长出来。
   ═══════════════════════════════════════════════ */
const cv = document.getElementById('field');
const ctx = cv.getContext('2d');
const GOLD = Math.PI * (3 - Math.sqrt(5));

let W, H, DPR;
function size(){
  DPR = Math.min(devicePixelRatio || 1, 2);
  W = innerWidth; H = innerHeight;
  cv.width = W*DPR; cv.height = H*DPR;
  cv.style.width = W+'px'; cv.style.height = H+'px';
  ctx.setTransform(DPR,0,0,DPR,0,0);
}
size(); addEventListener('resize', size);

const CAP = 260;
let bloom  = 0;        // 已长出的颗数
let target = 60;       // 目标颗数
let petalRGB = '255,255,255';
let petalA   = .34;

function readTheme(){
  const css = getComputedStyle(document.documentElement);
  petalA = parseFloat(css.getPropertyValue('--petal'));
}
readTheme();

/* 一颗水滴：尖端朝外 */
function drop(x, y, r, rot, alpha){
  ctx.save();
  ctx.translate(x, y); ctx.rotate(rot);
  ctx.beginPath();
  ctx.moveTo(0, -r*1.8);
  ctx.bezierCurveTo( r*1.06, -r*.46,  r*.98, r*.66, 0, r);
  ctx.bezierCurveTo(-r*.98,  r*.66, -r*1.06, -r*.46, 0, -r*1.8);
  ctx.closePath();
  const g = ctx.createLinearGradient(0, -r*1.8, 0, r);
  g.addColorStop(0, `rgba(${petalRGB},${alpha})`);
  g.addColorStop(1, `rgba(${petalRGB},${alpha*.3})`);
  ctx.fillStyle = g; ctx.fill();
  ctx.restore();
}

let t = 0;
function frame(){
  t += 0.0011;                                  // 极慢自转
  bloom += (target - bloom) * 0.010;            // 一颗一颗地长
  readTheme();

  ctx.clearRect(0,0,W,H);
  const cx = W/2, cy = H*0.43;
  const breath = 1 + Math.sin(t*1.9)*0.045;     // 缓慢呼吸
  const scale  = Math.min(W,H) * 0.075 * breath;
  const n = Math.ceil(bloom);

  for (let i=0; i<n; i++){
    const a   = i*GOLD + t;
    const rad = scale * Math.sqrt(i) * 0.86;
    const x = cx + Math.cos(a)*rad;
    const y = cy + Math.sin(a)*rad*0.88;

    const edge = 1 - i/CAP;                     // 外圈更淡
    const soft = Math.min(1, (bloom - i)/10);   // 新的一颗，慢慢淡入
    const r = (5.5 + Math.sqrt(i)*0.72) * (.5 + edge*.5);
    if (x < -40 || x > W+40 || y < -40 || y > H+40) continue;

    drop(x, y, r, a + Math.PI/2, petalA * edge * soft * 0.55);
  }
  requestAnimationFrame(frame);
}
if (!matchMedia('(prefers-reduced-motion: reduce)').matches) frame();
else bloom = target;

const root  = document.documentElement;
const sheet = document.getElementById('sheet');
const tgTheme = document.getElementById('tgTheme');

document.getElementById('gear').onclick = () => sheet.dataset.open = '1';
sheet.onclick = e => { if (e.target === sheet) sheet.dataset.open = '0'; };
tgTheme.onclick = () => {
  const light = root.dataset.theme === 'light';
  root.dataset.theme = light ? 'dark' : 'light';
  petalRGB = light ? '255,255,255' : '255,255,255';
  tgTheme.setAttribute('aria-checked', String(!light));
};

/* ═══ 对话（演示回复；接进项目时换成真实 API）═══ */
const stream = document.getElementById('stream');
const input  = document.getElementById('input');

function say(text, who){
  const d = document.createElement('div');
  d.className = 'msg ' + who; d.textContent = text;
  stream.appendChild(d); stream.scrollTop = stream.scrollHeight;
}
const demo = ['欸，来了。今天咋样',
              '还行是还行，还是那种"其实不太行但懒得说"的还行 🤔',
              '那先不管为什么。就说说今天最烦的那一件',
              '……我是不是问太急了',
              '我在'];
let k = 0;
say(demo[k++], 'ai');

document.getElementById('send').onclick = send;
input.onkeydown = e => { if (e.key === 'Enter') send(); };

function send(){
  const v = input.value.trim(); if (!v) return;
  say(v,'me'); input.value = '';
  target = Math.min(CAP, target + 20);          // 每轮对话，多开一圈

  const wait = document.createElement('div');
  wait.className = 'msg ai';
  wait.innerHTML = '<div class="dots"><i></i><i></i><i></i></div>';
  stream.appendChild(wait); stream.scrollTop = stream.scrollHeight;

  setTimeout(() => { wait.remove(); say(demo[k++ % demo.length], 'ai'); target = Math.min(CAP, target + 14); }, 900);
}
