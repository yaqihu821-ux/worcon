<script>
const cv  = document.getElementById('field');
const ctx = cv.getContext('2d');
const bg  = document.createElement('canvas');
const bx  = bg.getContext('2d');
const GOLD = Math.PI * (3 - Math.sqrt(5));

let W, H, DPR;
function size(){
  DPR = Math.min(devicePixelRatio || 1, 2);
  W = innerWidth; H = innerHeight;
  cv.width = W*DPR; cv.height = H*DPR;
  cv.style.width = W+'px'; cv.style.height = H+'px';
  ctx.setTransform(DPR,0,0,DPR,0,0);
  bg.width = W; bg.height = H;
}
size(); addEventListener('resize', size);

/* 噪点：平滑渐变没东西可折射，得给珠子一点"抓手" */
const noise = document.createElement('canvas');
noise.width = noise.height = 220;
(function(){
  const nx = noise.getContext('2d');
  const img = nx.createImageData(220, 220);
  for (let i = 0; i < img.data.length; i += 4){
    const v = 128 + (Math.random() - .5) * 90;
    img.data[i] = img.data[i+1] = img.data[i+2] = v;
    img.data[i+3] = 255;
  }
  nx.putImageData(img, 0, 0);
})();
const noisePat = bx.createPattern(noise, 'repeat');

const BREATH_W = 11;
const BREATH_A = 0.15;
const WAVE     = 0.42;

const CAP = 420;
let bloom = 0, target = 140;
let RIM = .5;
const isLight = () => document.documentElement.dataset.theme === 'light';

/* ─── 光场：茶苗青绿 × 浅雾蓝 ───
   "雾"靠低饱和实现，不是靠调低透明度。
   h 三个数 = 色相 / 饱和 / 亮度 */
const BLOBS = [
  { h:'142,46%,54%', ox:.24, oy:.28, fx:.000082, fy:.000061, px:0,   py:1.7, r:.68, amp:.26 }, // 嫩芽绿
  { h:'204,36%,60%', ox:.74, oy:.30, fx:.000054, fy:.000093, px:2.1, py:.4,  r:.66, amp:.30 }, // 浅雾蓝
  { h:'162,42%,50%', ox:.54, oy:.72, fx:.000071, fy:.000047, px:4.0, py:2.9, r:.62, amp:.24 }, // 茶青
  { h:'196,30%,52%', ox:.22, oy:.76, fx:.000039, fy:.000078, px:1.2, py:5.1, r:.58, amp:.32 }, // 雾蓝偏灰
];
const BASE_DARK  = '#05090B';
const BASE_LIGHT = '#C4D3CE';

function paintField(now){
  const light = isLight();
  bx.globalCompositeOperation = 'source-over';
  bx.fillStyle = light ? BASE_LIGHT : BASE_DARK;
  bx.fillRect(0,0,W,H);

  bx.globalCompositeOperation = light ? 'multiply' : 'lighter';
  const strength = light ? .52 : .60;

  for (const b of BLOBS){
    const x = W * (b.ox + Math.sin(now*b.fx + b.px) * b.amp);
    const y = H * (b.oy + Math.cos(now*b.fy + b.py) * b.amp * .8);
    const r = Math.max(W,H) * b.r * (1 + Math.sin(now*b.fx*.6)*.10);
    const g = bx.createRadialGradient(x,y,0,x,y,r);
    g.addColorStop(0,   `hsla(${b.h},${strength})`);
    g.addColorStop(.45, `hsla(${b.h},${strength*.36})`);
    g.addColorStop(1,   `hsla(${b.h},0)`);
    bx.fillStyle = g;
    bx.beginPath(); bx.arc(x,y,r,0,Math.PI*2); bx.fill();
  }

  bx.globalCompositeOperation = 'overlay';
  bx.globalAlpha = light ? .05 : .085;
  bx.fillStyle = noisePat;
  bx.fillRect(0,0,W,H);
  bx.globalAlpha = 1;

  bx.globalCompositeOperation = 'source-over';
  const v = bx.createRadialGradient(W/2, H*.44, Math.min(W,H)*.26, W/2, H*.44, Math.max(W,H)*.80);
  v.addColorStop(0, 'rgba(0,0,0,0)');
  v.addColorStop(1, light ? 'rgba(30,52,46,.34)' : 'rgba(0,0,0,.62)');
  bx.fillStyle = v; bx.fillRect(0,0,W,H);
}

/* ─── 真正的玻璃折射 ───────────────────────────
   均匀放大看起来是塑料。真玻璃是：中心几乎不变形，
   越靠边缘弯折越厉害、把周围的景象挤成极薄的一圈。
   做法：把珠子切成若干同心环带，每一环用不同的
   采样比例去取背景 —— 边缘的环压缩得极狠。
   K 越大，边缘挤压越强。
   ──────────────────────────────────────────── */
const K     = 1.05;   // 折射强度
const RINGS = 5;      // 环带数：越多越平滑，也越吃性能

function lens(u){ return u / (1 + K*u*u); }   // 源半径 ← 目标半径

function bead(x, y, r, a, shine){
  ctx.save();
  ctx.globalAlpha = a;

  for (let k = RINGS; k >= 1; k--){
    const uOut = k / RINGS;
    const uIn  = (k-1) / RINGS;
    const uMid = (uOut + uIn) / 2;
    const scl  = lens(uMid) / uMid;            // 这一环的采样缩放

    ctx.save();
    ctx.beginPath();
    ctx.arc(x, y, r*uOut, 0, Math.PI*2);
    if (uIn > 0) ctx.arc(x, y, r*uIn, 0, Math.PI*2, true);   // 挖空内圈
    ctx.clip('evenodd');

    const S = r * scl;
    ctx.drawImage(bg, x-S, y-S, S*2, S*2, x-r, y-r, r*2, r*2);
    ctx.restore();
  }
  ctx.restore();

  // 边缘那圈亮光：左上亮，右下淡
  const R = RIM * a * shine;
  const g = ctx.createLinearGradient(x-r, y-r, x+r, y+r);
  g.addColorStop(0,   `rgba(255,255,255,${R})`);
  g.addColorStop(.45, `rgba(255,255,255,${R*.16})`);
  g.addColorStop(.68, `rgba(255,255,255,${R*.05})`);
  g.addColorStop(1,   `rgba(255,255,255,${R*.45})`);
  ctx.beginPath(); ctx.arc(x, y, r*.985, 0, Math.PI*2);
  ctx.strokeStyle = g; ctx.lineWidth = Math.max(.8, r*.075); ctx.stroke();

  // 极淡的色散：玻璃边缘会把光分成冷暖两侧
  ctx.globalCompositeOperation = 'screen';
  ctx.beginPath(); ctx.arc(x, y, r*.96, Math.PI*.85, Math.PI*1.55);
  ctx.strokeStyle = `rgba(120,220,255,${R*.30})`;
  ctx.lineWidth = Math.max(.6, r*.05); ctx.stroke();
  ctx.beginPath(); ctx.arc(x, y, r*.96, Math.PI*.05, Math.PI*.55);
  ctx.strokeStyle = `rgba(255,215,160,${R*.22})`;
  ctx.lineWidth = Math.max(.6, r*.05); ctx.stroke();
  ctx.globalCompositeOperation = 'source-over';
}

function breathe(p){
  const s = Math.sin(p);
  return Math.sign(s) * Math.pow(Math.abs(s), 0.72);
}

let sec = 0, last = 0;
function frame(now){
  if (!last) last = now;
  sec += Math.min(.05, (now - last) / 1000);
  last = now;

  bloom += (target - bloom) * .010;
  RIM = parseFloat(getComputedStyle(document.documentElement).getPropertyValue('--rim'));

  paintField(now);
  ctx.clearRect(0,0,W,H);
  ctx.drawImage(bg, 0, 0, W, H);

  const cx = W/2, cy = H*.43;
  const spin = sec * .017;
  const w = Math.PI * 2 / BREATH_W;
  const scale = Math.min(W,H) * .056;
  const n = Math.ceil(bloom);

  for (let i=0; i<n; i++){
    const dist = Math.sqrt(i);
    const b    = breathe(sec*w - dist*WAVE);

    const ang = i*GOLD + spin;
    const rad = scale * dist * .88 * (1 + b * BREATH_A);
    const x = cx + Math.cos(ang)*rad;
    const y = cy + Math.sin(ang)*rad*.88;

    const edge = 1 - i/CAP;
    const soft = Math.min(1, (bloom-i)/10);
    const r     = (7.2 + dist*.62) * (.58 + edge*.42) * (1 + b*.14);
    const shine = .72 + b * .28;

    if (x < -30 || x > W+30 || y < -30 || y > H+30) continue;
    bead(x, y, r, Math.min(1, edge*soft*1.1), shine);
  }
  requestAnimationFrame(frame);
}
if (!matchMedia('(prefers-reduced-motion: reduce)').matches) requestAnimationFrame(frame);
else { paintField(0); ctx.drawImage(bg,0,0,W,H); }

  tgTheme.setAttribute('aria-checked', String(!light));

}
const demo = ['欸，来了。今天咋样',
              '还行是还行，还是那种"其实不太行但懒得说"的还行 🤔',

input.onkeydown = e => { if (e.key === 'Enter') send(); };

function send(){
  const v = input.value.trim(); if (!v) return;
  say(v,'me'); input.value = '';
  target = Math.min(CAP, target + 26);

  const wait = document.createElement('div');
  wait.className = 'msg lg ai';
  wait.innerHTML = '<div class="dots"><i></i><i></i><i></i></div>';
  stream.appendChild(wait); stream.scrollTop = stream.scrollHeight;

  setTimeout(() => { wait.remove(); say(demo[k++ % demo.length], 'ai'); target = Math.min(CAP, target + 18); }, 900);
}
