<script>
/* ══ 签名元素 ══════════════════════════════════════
   向日葵排列（叶序 / phyllotaxis）：第 n 颗种子的角度
   是黄金角 137.5° 的 n 倍，半径是 √n。自然界里向日葵、
   松果、菠萝都长这样。每完成一轮问答，花心多开一圈。
   ══════════════════════════════════════════════ */

const cv = document.getElementById('field');
const ctx = cv.getContext('2d');
const GOLD = Math.PI * (3 - Math.sqrt(5));   // 黄金角 ≈ 137.5°

let W, H, DPR;
function size(){
  DPR = Math.min(devicePixelRatio || 1, 2);
  W = innerWidth; H = innerHeight;
  cv.width = W * DPR; cv.height = H * DPR;
  cv.style.width = W + 'px'; cv.style.height = H + 'px';
  ctx.setTransform(DPR, 0, 0, DPR, 0, 0);
}
size();
addEventListener('resize', size);

let bloom = 0;        // 当前实际绽放量
let target = 26;      // 目标绽放量（随对话增长）
const CAP = 320;

/* 一颗水滴：尖端朝向花心外侧 */
function drop(x, y, r, rot, alpha){
  ctx.save();
  ctx.translate(x, y);
  ctx.rotate(rot);
  ctx.beginPath();
  ctx.moveTo(0, -r * 1.75);                          // 尖
  ctx.bezierCurveTo( r * 1.05, -r * .5,  r, r * .62, 0, r);   // 右腹
  ctx.bezierCurveTo(-r, r * .62, -r * 1.05, -r * .5, 0, -r * 1.75); // 左腹
  ctx.closePath();
  const seed = getComputedStyle(document.documentElement).getPropertyValue('--seed').trim();
  const a = parseFloat(getComputedStyle(document.documentElement).getPropertyValue('--seed-a'));
  const g = ctx.createLinearGradient(0, -r * 1.75, 0, r);
  g.addColorStop(0, `rgba(${seed},${alpha * a})`);
  g.addColorStop(1, `rgba(${seed},${alpha * a * .35})`);
  ctx.fillStyle = g;
  ctx.fill();
  ctx.restore();
}

let t = 0;
function frame(){
  t += 0.0016;
  bloom += (target - bloom) * 0.022;              // 缓缓散开

  ctx.clearRect(0, 0, W, H);
  const cx = W / 2, cy = H * 0.44;
  const scale = Math.min(W, H) * 0.052;
  const n = Math.ceil(bloom);

  for (let i = 0; i < n; i++){
    const ang = i * GOLD + t;                     // 极慢自转
    const rad = scale * Math.sqrt(i) * 0.92;
    const x = cx + Math.cos(ang) * rad;
    const y = cy + Math.sin(ang) * rad * 0.86;    // 轻微压扁，像看向水面
    if (x < -60 || x > W + 60 || y < -60 || y > H + 60) continue;

    const edge = 1 - i / CAP;                     // 外圈更淡
    const soft = Math.min(1, (bloom - i) / 6);    // 新种子淡入
    const r = (3.4 + Math.sqrt(i) * .42) * (.5 + edge * .5);
    drop(x, y, r, ang + Math.PI / 2, 0.30 * edge * soft);
  }
  requestAnimationFrame(frame);
}

if (!matchMedia('(prefers-reduced-motion: reduce)').matches) frame();
else { bloom = target; ctx.clearRect(0,0,W,H); }

  const light = root.dataset.theme === 'light';
  root.dataset.theme = light ? 'dark' : 'light';
  tg.setAttribute('aria-checked', String(!light));
};

/* ══ 对话（这里是演示回复；接进项目时换成 API）══ */
const stream = document.getElementById('stream');
const input  = document.getElementById('input');

function say(text, who){
  const d = document.createElement('div');
  d.className = 'msg ' + who;
  d.textContent = text;
  stream.appendChild(d);
  stream.scrollTop = stream.scrollHeight;
}

const demo = [
  '欸，来了。今天咋样',
  '还行是还行，还是那种"其实不太行但懒得说"的还行 🤔',
  '那先不管为什么。就说说今天最烦的那一件',
  '……我是不是问太急了',
  '我在'
];
let k = 0;
say(demo[k++], 'ai');

document.getElementById('send').onclick = send;
input.onkeydown = e => { if (e.key === 'Enter') send(); };

function send(){
  const v = input.value.trim();
  if (!v) return;
  say(v, 'me');
  input.value = '';
  target = Math.min(CAP, target + 22);            // 花开一圈

  const wait = document.createElement('div');
  wait.className = 'msg ai';
  wait.innerHTML = '<div class="dots"><i></i><i></i><i></i></div>';
  stream.appendChild(wait);
  stream.scrollTop = stream.scrollHeight;

  setTimeout(() => {
    wait.remove();
    say(demo[k++ % demo.length], 'ai');
    target = Math.min(CAP, target + 16);
  }, 900);
}
