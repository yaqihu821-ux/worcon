<script>
const cv  = document.getElementById('field');
const ctx = cv.getContext('2d');
const bg  = document.createElement('canvas');
const bx  = bg.getContext('2d');
const GOLD = Math.PI * (3 - Math.sqrt(5));

let W, H, DPR;
function size(){
  DPR = Math.min(devicePixelRatio || 1, 2);
  W = innerWidth; H = innerHeight;
  cv.width = W*DPR; cv.height = H*DPR;
  cv.style.width = W+'px'; cv.style.height = H+'px';
  ctx.setTransform(DPR,0,0,DPR,0,0);
  bg.width = W; bg.height = H;
}
size(); addEventListener('resize', size);

const CAP = 420;
let bloom = 0, target = 170;
let RIM = .55;
const isLight = () => document.documentElement.dataset.theme === 'light';

/* ─── 呼吸参数 ─────────────────────────────
   BREATH_W  周期（秒）—— 越大越慢
   BREATH_A  幅度 —— 越大起伏越深
   WAVE      从花心向外传播的延迟，0 就是齐步走
   ──────────────────────────────────────── */
const BREATH_W = 11;
const BREATH_A = 0.15;
const WAVE     = 0.42;

const BLOBS = [
  { h:'232,92%,42%', ox:.26, oy:.30, fx:.000082, fy:.000061, px:0,   py:1.7, r:.66, amp:.26 },
  { h:'270,85%,44%', ox:.72, oy:.26, fx:.000054, fy:.000093, px:2.1, py:.4,  r:.60, amp:.30 },
  { h:'198,92%,40%', ox:.56, oy:.70, fx:.000071, fy:.000047, px:4.0, py:2.9, r:.70, amp:.24 },
  { h:'322,78%,46%', ox:.20, oy:.76, fx:.000039, fy:.000078, px:1.2, py:5.1, r:.54, amp:.32 },
];
const BASE_DARK  = '#030407';
const BASE_LIGHT = '#D6DBE4';

function paintField(now){
  const light = isLight();
  bx.globalCompositeOperation = 'source-over';
  bx.fillStyle = light ? BASE_LIGHT : BASE_DARK;
  bx.fillRect(0,0,W,H);

  bx.globalCompositeOperation = light ? 'multiply' : 'lighter';
  const strength = light ? .46 : .92;

  for (const b of BLOBS){
    const x = W * (b.ox + Math.sin(now*b.fx + b.px) * b.amp);
    const y = H * (b.oy + Math.cos(now*b.fy + b.py) * b.amp * .8);
    const r = Math.max(W,H) * b.r * (1 + Math.sin(now*b.fx*.6)*.10);
    const g = bx.createRadialGradient(x,y,0,x,y,r);
    g.addColorStop(0,   `hsla(${b.h},${strength})`);
    g.addColorStop(.45, `hsla(${b.h},${strength*.38})`);
    g.addColorStop(1,   `hsla(${b.h},0)`);
    bx.fillStyle = g;
    bx.beginPath(); bx.arc(x,y,r,0,Math.PI*2); bx.fill();
  }

  bx.globalCompositeOperation = 'source-over';
  const v = bx.createRadialGradient(W/2, H*.44, Math.min(W,H)*.30, W/2, H*.44, Math.max(W,H)*.78);
  v.addColorStop(0, 'rgba(0,0,0,0)');
  v.addColorStop(1, light ? 'rgba(60,66,80,.16)' : 'rgba(0,0,0,.55)');
  bx.fillStyle = v; bx.fillRect(0,0,W,H);
}

const MAG = 1.45;
function bead(x, y, r, a, shine){
  const s = r / MAG;
  ctx.save();
  ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI*2); ctx.clip();
  ctx.globalAlpha = a;
  ctx.drawImage(bg, x-s, y-s, s*2, s*2, x-r, y-r, r*2, r*2);
  ctx.restore();

  const R = RIM * a * shine;
  const g = ctx.createLinearGradient(x-r, y-r, x+r, y+r);
  g.addColorStop(0,   `rgba(255,255,255,${R})`);
  g.addColorStop(.45, `rgba(255,255,255,${R*.18})`);
  g.addColorStop(.68, `rgba(255,255,255,${R*.06})`);
  g.addColorStop(1,   `rgba(255,255,255,${R*.42})`);
  ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI*2);
  ctx.strokeStyle = g; ctx.lineWidth = Math.max(.7, r*.09); ctx.stroke();
}

/* 真实的呼吸不是正弦：吸气慢、顶点停一下、呼气略快。
   把正弦拉一下形状，让它有"憋住"的那一刻。          */
function breathe(phase){
  const s = Math.sin(phase);
  return Math.sign(s) * Math.pow(Math.abs(s), 0.72);
}

let sec = 0, last = 0;
function frame(now){
  if (!last) last = now;
  sec += Math.min(.05, (now - last) / 1000);   // 真实秒数，跟帧率无关
  last = now;

  bloom += (target - bloom) * .010;
  RIM = parseFloat(getComputedStyle(document.documentElement).getPropertyValue('--rim'));

  paintField(now);
  ctx.clearRect(0,0,W,H);
  ctx.drawImage(bg, 0, 0, W, H);

  const cx = W/2, cy = H*.43;
  const spin = sec * .017;                      // 极慢自转
  const w = Math.PI * 2 / BREATH_W;             // 呼吸角速度
  const scale = Math.min(W,H) * .049;
  const n = Math.ceil(bloom);

  for (let i=0; i<n; i++){
    /* 关键：相位随离花心的距离递增 →
       一次吸气是从中心向外荡开的一圈，而不是整片一起动 */
    const dist  = Math.sqrt(i);
    const b     = breathe(sec*w - dist*WAVE);

    const ang = i*GOLD + spin;
    const rad = scale * dist * .88 * (1 + b * BREATH_A);      // ← 整体扩张收缩
    const x = cx + Math.cos(ang)*rad;
    const y = cy + Math.sin(ang)*rad*.88;

    const edge = 1 - i/CAP;
    const soft = Math.min(1, (bloom-i)/10);

    const r     = (4.8 + dist*.46) * (.55 + edge*.45) * (1 + b*.14);
    const shine = .72 + b * .28;                               // 吸气时更亮

    if (x < -30 || x > W+30 || y < -30 || y > H+30) continue;
    bead(x, y, r, Math.min(1, edge*soft*1.1), shine);
  }
  requestAnimationFrame(frame);
}
if (!matchMedia('(prefers-reduced-motion: reduce)').matches) requestAnimationFrame(frame);
else { paintField(0); ctx.drawImage(bg,0,0,W,H); }

const root = document.documentElement;
const sheet = document.getElementById('sheet');
const tgTheme = document.getElementById('tgTheme');

document.getElementById('gear').onclick = () => sheet.dataset.open = '1';
sheet.onclick = e => { if (e.target === sheet) sheet.dataset.open = '0'; };
tgTheme.onclick = () => {
  const light = isLight();
  root.dataset.theme = light ? 'dark' : 'light';
  tgTheme.setAttribute('aria-checked', String(!light));
};

const stream = document.getElementById('stream');
const input  = document.getElementById('input');

function say(text, who){
  const d = document.createElement('div');
  d.className = 'msg lg ' + who; d.textContent = text;
  stream.appendChild(d); stream.scrollTop = stream.scrollHeight;
}
const demo = ['欸，来了。今天咋样',
              '还行是还行，还是那种"其实不太行但懒得说"的还行 🤔',
              '那先不管为什么。就说说今天最烦的那一件',
              '……我是不是问太急了',
              '我在'];
let k = 0;
say(demo[k++], 'ai');

document.getElementById('send').onclick = send;
input.onkeydown = e => { if (e.key === 'Enter') send(); };

function send(){
  const v = input.value.trim(); if (!v) return;
  say(v,'me'); input.value = '';
  target = Math.min(CAP, target + 26);

  const wait = document.createElement('div');
  wait.className = 'msg lg ai';
  wait.innerHTML = '<div class="dots"><i></i><i></i><i></i></div>';
  stream.appendChild(wait); stream.scrollTop = stream.scrollHeight;

  setTimeout(() => { wait.remove(); say(demo[k++ % demo.length], 'ai'); target = Math.min(CAP, target + 18); }, 900);
}
