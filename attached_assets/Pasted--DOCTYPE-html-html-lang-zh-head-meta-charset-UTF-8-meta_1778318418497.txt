/* ══════════════════════════════════
   LIQUID LIGHT — Perlin-style noise
══════════════════════════════════ */
const cv = document.getElementById('bg');
const cx = cv.getContext('2d');
let W, H;
const resize = () => { W = cv.width = innerWidth; H = cv.height = innerHeight; };
resize(); window.addEventListener('resize', resize);

// Simple smooth noise via sin harmonics
function noise(x, y, t) {
  return (
    Math.sin(x * 1.2 + t * 0.4) * Math.cos(y * 0.9 + t * 0.3) * 0.4 +
    Math.sin(x * 2.3 + y * 1.7 + t * 0.6) * 0.3 +
    Math.cos(x * 0.7 - y * 2.1 + t * 0.5) * 0.3
  );
}

const COLORS = [
  [92,  111, 255],  // blue-violet
  [220,  80, 255],  // magenta
  [60,  220, 200],  // teal
  [255,  80, 160],  // pink
  [100, 180, 255],  // sky
];

let frame = 0;
// We draw blobs whose centers oscillate using noise
const blobs = Array.from({length:6}, (_,i) => ({
  baseX: Math.random(), baseY: Math.random(),
  phase: Math.random() * Math.PI * 2,
  speed: 0.0003 + Math.random() * 0.0003,
  color: COLORS[i % COLORS.length],
  size: 0.28 + Math.random() * 0.18,
}));

function draw() {
  cx.clearRect(0, 0, W, H);
  const t = frame * 0.4;

  blobs.forEach(b => {
    const nx = noise(b.baseX * 3, b.baseY * 3, t * b.speed * 800 + b.phase);
    const ny = noise(b.baseX * 3 + 100, b.baseY * 3 + 100, t * b.speed * 800 + b.phase + 2);
    const px = (b.baseX + nx * 0.22) * W;
    const py = (b.baseY + ny * 0.22) * H;
    const r  = b.size * Math.min(W, H);
    const [r0, g0, b0] = b.color;

    // Pulsing size
    const rr = r * (0.88 + 0.12 * Math.sin(t * 0.008 + b.phase));

    const grad = cx.createRadialGradient(px, py, 0, px, py, rr);
    grad.addColorStop(0,   `rgba(${r0},${g0},${b0},0.14)`);
    grad.addColorStop(0.45,`rgba(${r0},${g0},${b0},0.06)`);
    grad.addColorStop(1,   `rgba(${r0},${g0},${b0},0)`);

    cx.beginPath();
    cx.arc(px, py, rr, 0, Math.PI * 2);
    cx.fillStyle = grad;
    cx.fill();
  });

  frame++;
  requestAnimationFrame(draw);
}
draw();

function emotion(t) {
  if (/难过|伤心|哭|泪|失去|孤独|想念|心碎|绝望|痛苦|委屈/.test(t)) return 'sad';
  if (/愤怒|生气|烦|讨厌|恨|气死|崩溃|受够/.test(t)) return 'angry';
  if (/焦虑|紧张|担心|害怕|恐惧|不安|压力|焦躁|慌/.test(t)) return 'anxious';
  if (/开心|快乐|高兴|喜欢|感谢|幸福|太棒|满足|兴奋/.test(t)) return 'happy';
  if (/喜欢你|爱你|心动|温暖|爱/.test(t)) return 'love';
  if (/迷茫|不知道|纠结|搞不懂|为什么|怎么办|困惑/.test(t)) return 'confused';
  return 'neutral';
}

const SYSTEM = `你是 Worcon，一个温柔、有深度的心理陪伴助手。风格：
- 安静、有温度，不说大道理
- 先倾听和共情，再轻轻回应

- 绝不给建议除非用户主动要求
- 只用中文，不用任何markdown格式`;

async function callClaude(msgs) {
  const res = await fetch('https://api.anthropic.com/v1/messages', {
    method:'POST',
    headers:{'Content-Type':'application/json'},
    body: JSON.stringify({ model:currentModel, max_tokens:1000, system:SYSTEM, messages:msgs })
  });
  const d = await res.json();
  return d.content?.[0]?.text || '我在这里。';
}

async function callOpenAI(msgs) {
  const messages = [{role:'system',content:SYSTEM}, ...msgs];
  const res = await fetch('https://api.openai.com/v1/chat/completions', {
    method:'POST',
    headers:{'Content-Type':'application/json','Authorization':`Bearer ${apiKey}`},
    body: JSON.stringify({ model:currentModel, messages, max_tokens:300 })
  });
  const d = await res.json();
  return d.choices?.[0]?.message?.content || '我在这里。';
}

async function callGemini(msgs) {
  const contents = msgs.map(m=>({role:m.role==='assistant'?'model':'user',parts:[{text:m.content}]}));
  const url = `https://generativelanguage.googleapis.com/v1beta/models/${currentModel}:generateContent?key=${apiKey}`;
  const res = await fetch(url, {
    method:'POST',
    headers:{'Content-Type':'application/json'},
    body: JSON.stringify({ contents, systemInstruction:{parts:[{text:SYSTEM}]} })
  });
  const d = await res.json();
  return d.candidates?.[0]?.content?.parts?.[0]?.text || '我在这里。';
}

async function send() {
  const text = inp.value.trim(); if (!text) return;
  inp.value=''; autoResize(inp);
  addUserMsg(text);
  history.push({role:'user', content:text});
  sbtn.disabled=true; showTyping();
  try {
    let reply;
    if      (currentProvider==='claude') reply = await callClaude(history);
    else if (currentProvider==='openai') reply = await callOpenAI(history);
    else                                  reply = await callGemini(history);
    rmTyping(); addAiMsg(reply);
    history.push({role:'assistant',content:reply});
  } catch(e) {
    rmTyping(); addAiMsg('我在这里，信号有点小问题。稍后再试？');
  }
  sbtn.disabled=false; inp.focus();
}
